# =====================================================================
#  Словники Radio 345 — шаблон
# =====================================================================
#
#  Як користуватися:
#    1. Скопіюйте цей файл:  cp dictionaries.example.txt dictionaries.txt
#    2. Впишіть свої записи у три розділи нижче (приклади — під #).
#    3. Запустіть приймач. Якщо база порожня, програма сама запропонує
#       імпортувати dictionaries.txt — відповідайте Y.
#       Вручну:  .venv/bin/python manage_database.py init
#
#  Після імпорту словники живуть у базі й редагуються в Mini App.
#  dictionaries.txt приватний і не потрапляє в Git.
#
#  Загальні правила:
#    • Розділ починається з рядка [назва]. Розділи: [callsigns], [commands],
#      [telegram]. Кожен розділ — не більше одного разу; порядок довільний.
#    • Один запис — один рядок. Назва — одне слово (апостроф і дефіс можна).
#    • Рядок, що починається з #, — коментар; він не імпортується.
#    • «# @radio345-disabled: запис» — вимкнений запис: зберігається в базі,
#      але не використовується, доки його не ввімкнуть у Mini App.
#    • Кодування — UTF-8.
#
#  Формати запису (однакові для всіх розділів):
#    слово                      назва без номера
#    слово: 68                  назва з одним номером
#    слово: 68 69 70            кілька номерів через пробіл
#    слово: 100-114             діапазон номерів
#    слово: 5 10 100-114        номери й діапазони разом
#    слово: *                   назва з будь-яким номером
#    слово 1313                 «слово номер» — лише один номер, без двокрапки
#
#  Для кількох номерів, діапазонів і * двокрапка обов’язкова.


# ---------------------------------------------------------------------
#  Позивні та дозволені номери. ОБОВ’ЯЗКОВО: хоча б один увімкнений запис,
#  інакше приймач не запуститься.
# ---------------------------------------------------------------------
[callsigns]
# Приклади:
# липка: 700 701 702
# наган: 100-114
# ліра: 5 10 17 68
# сапсан
# прибій: *


# ---------------------------------------------------------------------
#  Команди та ключові слова, які треба виділяти у фразі.
#  Розділ може бути порожнім — тоді виявлення команд вимкнено.
# ---------------------------------------------------------------------
[commands]
# Приклади:
# буревій 1313
# шторм 333
# штиль
# гроза


# ---------------------------------------------------------------------
#  Список спостереження для режиму Telegram «watch»: у цьому режимі
#  надсилаються лише фрази, де є хоча б один із цих записів.
#  Розділ може бути порожнім.
# ---------------------------------------------------------------------
[telegram]
# Приклади:
# ліра: 68
# буревій
# шторм
